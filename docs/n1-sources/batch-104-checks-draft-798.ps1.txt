$ErrorActionPreference = 'Stop'
$control = $PSScriptRoot
$root = "$control\..\learning-japanese"
$android = "$control\..\kanji-cards-android"
$session = 'a35853d9-7a7f-4820-9b94-71b73c4dd702'
$log = "$control\logs\$session"
Set-Location $root
$runner = [IO.File]::ReadAllText("$control\runner.json") | ConvertFrom-Json
if ($runner.status -ne 'running' -or $runner.pid -ne 13320 -or $runner.sessionId -ne $session) { throw 'Uncertain runner ownership' }
$process = Get-CimInstance Win32_Process -Filter "ProcessId=$PID"
for ($i = 0; $process -and $process.ProcessId -ne $runner.pid -and $i -lt 12; $i++) {
    $process = Get-CimInstance Win32_Process -Filter "ProcessId=$($process.ParentProcessId)"
}
if (-not $process -or $process.ProcessId -ne $runner.pid) { throw 'Live launcher ancestry missing' }
if ((git rev-parse HEAD).Trim() -ne 'b4b381728793f4c1d0d9b09f8eee47eaf5b8985e' -or
    (git log -1 '--format=%G?').Trim() -ne 'G') { throw 'Unexpected content signed head' }
if ((git -C $android rev-parse HEAD).Trim() -ne 'bf30cdead8515739bc0e671d57a54b9a3109dd46' -or
    (git -C $android log -1 '--format=%G?').Trim() -ne 'G' -or
    (git -C $android status --porcelain --untracked-files=all)) { throw 'Android not wholly clean at qualified signed head' }
$progress = [IO.File]::ReadAllText("$control\progress.json") | ConvertFrom-Json
if ($progress.checkpoint -ne 797 -or $progress.batch -ne '104' -or $progress.status -ne 'continue' -or
    $progress.stage -ne 'batch-104-runtime-draft-in-progress') { throw 'Unexpected checkpoint' }
$env:PYTHONPATH = "$root\tools"
python -X utf8 -B -m unittest test_freeze_n1_sources test_extract_n1_jmdict test_audit_n1_pilot test_audit_n1_batch *> "$log\source-tests-798.txt"
if ($LASTEXITCODE -ne 0) { throw 'Focused source tests failed' }
python -X utf8 -B tools\validate.py *> "$log\validate-798.txt"
if ($LASTEXITCODE -ne 0) { throw 'Validator failed' }
python -X utf8 -B tools\build_manifest.py *> "$log\manifest-798.txt"
if ($LASTEXITCODE -ne 0) { throw 'Manifest generation failed' }
python -X utf8 -B tools\audit_n1_batch.py --batch 104 *> "$log\batch-audit-798.txt"
if ($LASTEXITCODE -ne 0) { throw 'Existing batch audit failed' }
python -X utf8 -B "$control\draft-798.py" audit *> "$log\audit-798.txt"
if ($LASTEXITCODE -ne 0) { throw 'Exact draft/source/baseline/link/font audit failed' }
python -X utf8 -B tools\check_content_budget.py *> "$log\budget-798.txt"
if ($LASTEXITCODE -ne 0) { throw 'Publishing budgets failed' }
python -X utf8 -B tools\grammar_vocabulary.py check *> "$log\vocabulary-798.txt"
if ($LASTEXITCODE -ne 0) { throw 'Vocabulary consistency failed' }
git diff --check *> "$log\whitespace-798.txt"
if ($LASTEXITCODE -ne 0) { throw 'Whitespace check failed' }
Get-Content "$log\source-tests-798.txt" -Tail 4
Get-Content "$log\validate-798.txt" -Tail 2
Get-Content "$log\budget-798.txt" -Tail 8
Get-Content "$log\vocabulary-798.txt" -Tail 2
Write-Output 'Local UNREVIEWED draft checks passed; no app/native execution or publication.'
