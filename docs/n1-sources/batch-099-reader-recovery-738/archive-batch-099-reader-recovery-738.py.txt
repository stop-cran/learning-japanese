def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(path, value):
    if path.exists():
        raise RuntimeError(f"Refusing to overwrite evidence: {path}")
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def git(repo, *args):
    return subprocess.check_output(["git", "-C", str(repo), *args], text=True, encoding="utf-8").strip()


def update(before, after, allowed):
    for key, value in before.items():
        if key not in allowed and after.get(key) != value:
            raise RuntimeError(f"Historical value changed: {key}")
    target = CONTROL / "progress.json"
    if load(target) != before:
        raise RuntimeError("Progress changed during update")
    temporary = CONTROL / f"progress.json.738-{sys.argv[1]}.tmp"
    write(temporary, after)
    if load(temporary) != after:
        raise RuntimeError("Serialization mismatch")
    os.replace(temporary, target)
    if load(target) != after:
        raise RuntimeError("Persisted checkpoint mismatch")


def require(condition, message):
    if not condition:
        raise RuntimeError(message)
