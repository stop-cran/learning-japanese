$ErrorActionPreference = 'Stop'
$control = $PSScriptRoot
$root = "$control\..\learning-japanese"
$android = "$control\..\kanji-cards-android"
$session = '2b83c2d7-d1f1-4f27-9b9d-b5cd7a341d95'
$log = "$control\logs\$session"
Set-Location $root
$runner = [IO.File]::ReadAllText("$control\runner.json") | ConvertFrom-Json
if ($runner.status -ne 'running' -or $runner.pid -ne 13320 -or $runner.sessionId -ne $session) { throw 'Uncertain runner ownership' }
$process = Get-CimInstance Win32_Process -Filter "ProcessId=$PID"
for ($i = 0; $process -and $process.ProcessId -ne $runner.pid -and $i -lt 12; $i++) {
    $process = Get-CimInstance Win32_Process -Filter "ProcessId=$($process.ParentProcessId)"
}
if (-not $process -or $process.ProcessId -ne $runner.pid) { throw 'Live launcher ancestry missing' }
$preflight = [IO.File]::ReadAllText("$log\preflight-797.json") | ConvertFrom-Json
if ($preflight.lockTouched -or $preflight.sessionId -ne $session -or -not $preflight.decodedLauncherMatches) { throw 'Wrong preflight' }
if ((git rev-parse HEAD).Trim() -ne '9d44e63ffaa72875540f68cc55af284ffe6e4f56' -or
    (git log -1 '--format=%G?').Trim() -ne 'G') { throw 'Unexpected content signed head' }
if ((git -C $android rev-parse HEAD).Trim() -ne 'bf30cdead8515739bc0e671d57a54b9a3109dd46' -or
    (git -C $android log -1 '--format=%G?').Trim() -ne 'G' -or
    (git -C $android status --porcelain --untracked-files=all)) { throw 'Android not wholly clean at qualified signed head' }
$progress = [IO.File]::ReadAllText("$control\progress.json") | ConvertFrom-Json
if ($progress.checkpoint -ne 796 -or $progress.batch -ne '104' -or $progress.status -ne 'continue') { throw 'Unexpected checkpoint' }
foreach ($change in @(git status --porcelain --untracked-files=all)) {
    if ($change -notmatch '^( M|\?\?) docs/n1-sources/batch-104-') { throw "Unaccounted change: $change" }
}
$env:PYTHONPATH = "$root\tools"
python -X utf8 -B -m unittest test_freeze_n1_sources test_extract_n1_jmdict test_audit_n1_pilot test_audit_n1_batch *> "$log\source-tests-797.txt"
if ($LASTEXITCODE -ne 0) { throw 'Focused source tests failed' }
python -X utf8 -B tools\validate.py *> "$log\validate-797.txt"
if ($LASTEXITCODE -ne 0) { throw 'Validator failed' }
python -X utf8 -B tools\build_manifest.py *> "$log\manifest-797.txt"
if ($LASTEXITCODE -ne 0) { throw 'Manifest generation failed' }
git diff --exit-code -- manifest.json *> "$log\manifest-diff-797.txt"
if ($LASTEXITCODE -ne 0) { throw 'Source-only checkpoint changed runtime manifest' }
python -X utf8 -B tools\check_content_budget.py *> "$log\budget-797.txt"
if ($LASTEXITCODE -ne 0) { throw 'Publishing budgets failed' }
python -X utf8 -B tools\grammar_vocabulary.py check *> "$log\vocabulary-797.txt"
if ($LASTEXITCODE -ne 0) { throw 'Vocabulary consistency failed' }
python -X utf8 -B "$control\source-797.py" audit *> "$log\audit-797.txt"
if ($LASTEXITCODE -ne 0) { throw 'Source packet audit failed' }
git diff --check *> "$log\whitespace-797.txt"
if ($LASTEXITCODE -ne 0) { throw 'Whitespace check failed' }
Get-Content "$log\source-tests-797.txt" -Tail 4
Get-Content "$log\validate-797.txt" -Tail 2
Get-Content "$log\budget-797.txt" -Tail 8
Get-Content "$log\vocabulary-797.txt" -Tail 2
Get-Content "$log\audit-797.txt" -Tail 8
Write-Output 'Source-completion checks passed; no app/native execution or publication.'
