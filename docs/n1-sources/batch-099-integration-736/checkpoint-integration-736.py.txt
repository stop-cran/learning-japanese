def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def save(path, data):
    temp = path.with_suffix(path.suffix + ".736.tmp")
    assert not temp.exists(), temp
    temp.write_text(json.dumps(data, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(temp) == data
    temp.replace(path)


def git(*args, root=R):
    return subprocess.check_output(
        ["git", "--no-pager", "-C", str(root), *args], encoding="utf-8").strip()


def raw(commit, name):
    return subprocess.check_output(["git", "-C", str(R), "show", f"{commit}:{name}"])


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def audit():
    from audit_n1_batch import audit as batch_audit
    from audit_n1_pilot import metadata
    from build_manifest import build_manifest, sha256

    context()
    merge = git("rev-parse", "HEAD")
    assert git("show", "-s", "--format=%P") == f"{HEAD} {MAIN}"
    assert git("log", "-1", "--format=%G?") == "G"
    changed = git("diff", "--name-only", HEAD, merge).splitlines()
    assert changed == ["docs/content-format.md", "tools/test_validate.py", "tools/validate.py"]
    for name in changed:
        assert raw(merge, name) == raw(MAIN, name)
    old = raw(HEAD, "kanji/剰.md").decode("utf-8").replace("\r\n", "\n")
    now = (R / "kanji" / "剰.md").read_text(encoding="utf-8")
    assert now == old.replace("distractors: [乗, 剣, 則]\n",
                              "distractors: [乗, 剣, 則]\nquiz_exclusions: [冗]\n", 1)
    before = json.loads(raw(HEAD, "manifest.json"))
    current = load(R / "manifest.json")
    assert current == build_manifest(R)
    payloads = [(section, name, value) for section, entries in current.items()
                if isinstance(entries, dict) for name, value in entries.items()]
    delta = [(s, n) for s, n, h in payloads if before[s][n] != h]
    assert delta == [("files", "kanji/剰.md")]
    assert all(current[s].keys() == before[s].keys()
               for s, entries in current.items() if isinstance(entries, dict))
    for section, name, h in payloads:
        binary = section in ("images", "audio", "listeningTaskAudio", "listeningTaskImages")
        actual = digest(R / name) if binary else sha256(R / name)
        assert actual == h, name
    result = batch_audit(R, "099")
    assert (result["draftKanji"], result["newWords"], result["changedReusedArticles"],
            result["activatedBaselinePairs"], result["readingDispositions"],
            result["curatedOfferedKeys"]) == (1, 0, 2, 1, 4, 1)
    prior = metadata(R, "kanji/剰.md")
    assert prior["quiz_exclusions"] == ["冗"]
    word = metadata(R, "words/過剰.md")
    assert word["jlpt"] == 2 and word["reading"] == "かじょう"
    assert word["quiz_exclusions"] == load(S / "batch-099-draft-audit.json")["preservedWordExclusions"]
    pairs = [("剰", "過剰"), ("過", "過剰")]
    for char, spelling in pairs:
        assert f"(../words/{spelling}.md)" in (R / "kanji" / f"{char}.md").read_text(encoding="utf-8")
        assert f"(../kanji/{char}.md)" in (R / "words" / f"{spelling}.md").read_text(encoding="utf-8")
    assert git("-c", "core.quotepath=false", "diff", "--name-only").splitlines() == [
        "kanji/剰.md", "manifest.json"]
    assert not git("ls-files", "--others", "--exclude-standard")
    result.update({
        "schemaVersion": 1, "execution": 736, "integrationCommit": merge,
        "startingHead": HEAD, "integratedMain": MAIN,
        "integrationSignature": "G", "approvedToolDeltaExact": changed,
        "onlyChangedRuntimePayloadSince735": "kanji/剰.md",
        "exactSingleLineChangeVerified": True, "allOtherRuntimePayloadsUnchanged": True,
        "verifiedRuntimeHashes": len(payloads), "manifestSha256": digest(R / "manifest.json"),
        "allMappedReciprocalPairs": 2, "directedLinksVerified": 4,
        "preservedWordExclusions": word["quiz_exclusions"],
        "allSourcePacketsMembershipStrokesAndTeachingOtherwiseUnchanged": True,
        "kanjiExclusion": {"declaring": "剰", "exclude": ["冗"], "symmetricDirectNotTransitive": True,
                           "evidence": "batch-099-rendered-734-reconciliation.json N099-FALLBACK-JOU; complete frozen JMdict1196170/1355490/1595440 and KANJIDIC2 records",

    budget = ast.literal_eval((L / "budget-736.txt").read_text(encoding="utf-8-sig"))
    checks = {}

        checks[name] = {"log": str(source), "sha256": digest(source)}
    for name in ("validate", "manifest", "tests", "vocabulary", "budget", "whitespace"):
        assert load(L / "check-exits-736.json")[name] == 0
    state = load(L / "starting-state-736.json")
    evidence = {

        "budget": budget, "appMainPushCi": load(L / "app-main-ci-736.json"),
        "original733734Retroaccepted": False, "runtimeChanged": True,

        "freshAndroidOrNativeClaimed": False, "publishedAdditions": 125,
        "remainingAdditions": 1103, "lastPublishedCommit": p["lastPublishedCommit"],
        "physicalPhoneAcceptance": "outstanding", "generalReleaseAcceptance": "outstanding",
        "wholeProjectComplete": False, "nextBatchStarted": False,
        "parentLockTouched": False, "optionalHistoricalHelpersAccessed": False,
        "actualNewPermissionDenials": [], "nextAction": NEXT,
    }
    artifacts = S / "batch-099-integration-736"
    artifacts.mkdir(exist_ok=True)
    for name, info in checks.items():
        source = Path(info["log"])
        target = artifacts / f"{name}.txt.gz"
        target.write_bytes(gzip.compress(source.read_bytes(), mtime=0))

        info["gzipSha256"] = digest(target)
    for name in ("checkpoint-integration-736.py", "preflight-736.ps1", "checks-integration-736.ps1"):

    shutil.copyfile(L / "check-exits-736.json", artifacts / "check-exits.json")
    save(S / "execution-736.json", evidence)
    print("Evidence and exact compressed check logs recorded; not accepted/published.")
