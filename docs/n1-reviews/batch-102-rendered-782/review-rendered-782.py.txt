def load(path):
    return json.loads(path.read_bytes().decode("utf-8-sig"))


def save(path, value):
    assert not path.exists(), path
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    assert load(path) == value


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def now():
    return datetime.now(timezone.utc).isoformat()


def git(*args, root=R):
    return subprocess.check_output(["git", "--no-pager", "-C", str(root), *args],
                                   encoding="utf-8").strip()


def absent(pid):
    return subprocess.run(["pwsh", "-NoProfile", "-Command",
                           f"if (Get-Process -Id {int(pid)} -ErrorAction SilentlyContinue) {{ exit 1 }}"],
                          capture_output=True, check=False).returncode == 0

def atomic_progress(before, after, suffix):

def committed_bytes(name):
    entry = next(e for e in load(PRIOR / "artifact-index.json") if e["file"] == name)
    path = PRIOR / name
    assert sha(path) == entry["storedSha256"], name
    data = gzip.decompress(path.read_bytes()) if entry["losslessGzip"] else path.read_bytes()
    assert hashlib.sha256(data).hexdigest() == entry["sourceSha256"], name
    assert len(data) == entry["sourceBytes"]
    return data
