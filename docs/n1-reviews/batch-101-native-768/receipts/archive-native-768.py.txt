def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def save(path, data):
    assert not path.exists(), path
    path.write_text(json.dumps(data, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(path) == data


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git(*args, root=ROOT):
    return subprocess.check_output(
        ["git", "--no-pager", "-C", str(root), *args], encoding="utf-8"
    ).strip()


def verify_runtime():
    packet = load(CONTROL / "batch-101-native-767" / "compiled-packet.json")
    prep, compiled = packet["preparation"], packet["compilation"]
    for path, digest in [
        (prep["archive"], prep["archiveSha256"]),
        (prep["fixture"], prep["fixtureSha256"]),
        (compiled["appApk"], compiled["appApkSha256"]),
        (compiled["testApk"], compiled["testApkSha256"]),
    ]:
        assert sha(Path(path)) == digest
    manifest_bytes = (ROOT / "manifest.json").read_bytes()
    manifest = json.loads(manifest_bytes)
    assert manifest["contentVersion"] == "6e9b74967e21fc3c"
    assert hashlib.sha256(manifest_bytes).hexdigest() == prep["manifestSha256"]
    assert subprocess.check_output([
        "git", "-C", str(ROOT), "show", prep["contentSnapshot"] + ":manifest.json"
    ]) == manifest_bytes
    with zipfile.ZipFile(prep["archive"]) as archive:
        names = archive.namelist()
        prefix = "n1-batch-101/"
        assert len(names) == len(set(names)) == 8477 <= 18000
        assert all(n.startswith(prefix) and "\\" not in n and ".." not in n.split("/")
                   and not n.startswith("/") and not re.search(r"(^|/)[^/]*:", n) for n in names)
        assert not any("/docs/" in n or "/tools/" in n for n in names)
        assert archive.read(prefix + "manifest.json") == manifest_bytes
        retained = len(manifest_bytes)
        count = media = 0
        for section_name, section in manifest.items():
            if not isinstance(section, dict):
                continue
            for name, digest in section.items():
                assert len(name.split("/")) == 2 and ".." not in name.split("/")
                data = archive.read(prefix + name)
                local = (ROOT / name).read_bytes()
                binary = section_name in {"images", "audio", "listeningTaskAudio", "listeningTaskImages"}
                limit = 150 * 1024 if section_name in {"images", "listeningTaskImages"} else (
                    512 * 1024 if section_name in {"audio", "listeningTaskAudio"} else 2 * 1024 * 1024)
                assert len(data) <= limit, name
                normalized = data if binary else data.replace(b"\r\n", b"\n")
                assert hashlib.sha256(normalized).hexdigest() == digest, name
                assert normalized == (local if binary else local.replace(b"\r\n", b"\n")), name
                retained += len(data)
                count += 1
                if section_name in {"listeningTaskAudio", "listeningTaskImages"}:
                    media += len(data)
        inflated = sum(info.file_size for info in archive.infolist())
        compressed = Path(prep["archive"]).stat().st_size
        assert count == 8455 and retained == 44386225 <= 56 * 1024 * 1024
        assert inflated == 44654563 <= 80 * 1024 * 1024
        assert compressed == 30479587 <= 42 * 1024 * 1024
        assert media == 5893602 <= 20 * 1024 * 1024
    return {
        "runtimeHashesVerified": count, "contentVersion": manifest["contentVersion"],
        "manifestSha256": sha(ROOT / "manifest.json"), "manifestEqualsCompiledArchive": True,
        "archiveContentSnapshot": prep["contentSnapshot"], "archiveSha256": prep["archiveSha256"],
        "zipEntries": len(names), "compressedBytes": compressed, "inflatedBytes": inflated,
        "retainedBytes": retained, "listeningTaskMediaBytes": media,
        "allOriginalBudgetsAndPerFilePathMediaGuardsPassed": True,
    }


def verify_native():

    assert attempt["coordinatorHead"] == BASE and attempt["androidSnapshot"] == APP
    text = (RAW / "instrumentation.txt").read_text(encoding="utf-8-sig")
    packet = load(CONTROL / "batch-101-native-767" / "compiled-packet.json")["preparation"]
    assert "OK (2 tests)" in text and text.count("INSTRUMENTATION_STATUS_CODE: 0") == 2
    assert not re.search(r"INSTRUMENTATION_STATUS_CODE: -[234]|FAILURES|INSTRUMENTATION_FAILED|INSTRUMENTATION_STATUS: (?:ignored|assumptionFailure)", text)
    for key in ("glyphClass", "articleClass"):
        assert text.count("INSTRUMENTATION_STATUS: class=" + packet[key]) == 2, key
    for key in ("glyphMethod", "articleMethod"):
        assert text.count("INSTRUMENTATION_STATUS: test=" + packet[key]) == 2, key
    route = load(RAW / "n1-101-article-route.json")
    raster = load(RAW / "n1-101-font-raster.json")
    assert route["passed"] and route["archiveRequests"] == 1
    assert (route["importedKanji"], route["importedWords"], route["n1Cards"], route["problems"],
            route["acceptedReadingParserEpoch"], route["acceptedMeaningParserEpoch"]) == (1124, 5073, 133, 0, 1, 1)
    assert route["contentSnapshot"] == packet["contentSnapshot"]
    assert route["contentVersion"] == packet["contentVersion"]
    assert route["fullOnyomi"] == route["fullProjectionOnyomi"] == ["コウ", "ク"]
    assert route["fullKunyomi"] == route["fullProjectionKunyomi"] == ["いさお"]
    assert route["quizOnKeys"] == ["こう"] and route["quizKunKeys"] == []
    assert route["fullProjectionQuizOnyomi"] == ["コウ"] and route["fullProjectionQuizKunyomi"] == []
    guards = ["績", "効", "果", "成", "達", "勝", "凱", "栄", "敬", "働", "業", "技", "工"]
    assert route["kanjiQuizExclusions"] == guards
    rows = route["kanjiDirectGuardChecks"]
    assert len(rows) == 2912 and len({r["pool"] for r in rows}) == 7
    for guard in guards:
        selected = [r for r in rows if {r["target"], r["excluded"]} == {"功", guard}]
        assert len(selected) == 224
        assert {r["target"] for r in selected} == {"功", guard}
        assert {r["seed"] for r in selected} == set(range(16))
        for row in selected:
            assert row["target"] in row["options"] and row["excluded"] not in row["options"]
            assert 1 <= len(row["options"]) <= 4 and len(row["options"]) == len(set(row["options"]))
            assert (set(row["options"]) - {row["target"]}).issubset(set(row["poolMembershipInOrder"]))
    nontransitive = route["nontransitiveNeighborChecks"]
    assert len(nontransitive) == 2496
    assert len({(r["target"], r["neighbor"]) for r in nontransitive}) == 156
    assert all(set(r["options"]) == {r["target"], r["neighbor"]} for r in nontransitive)
    word_checks = route["preservedWordGuardChecks"]
    assert len(word_checks) == 896
    assert len({tuple(sorted((r["target"], r["excluded"]))) for r in word_checks}) == 7
    for row in word_checks:
        assert row["target"] in {o["word"] for o in row["options"]}
        assert row["excluded"] not in {o["word"] for o in row["options"]}
        assert len(row["options"]) == 4
    choices = route["productionBuilderN1AndAllExactChoices"]
    assert len(choices) == 64
    expected_meanings = {"功": "merit; achievement", "加": "add",
                         "幼": "early childhood; immature", "刀": "sword; blade"}
    for row in choices:
        ids = {o["kanji"] for o in row["options"]}
        assert len(ids) == len(row["options"]) == 4 and "功" in ids
        assert not ids.intersection(guards) and ids.issubset(set(row["poolMembershipInOrder"]))
        if row["pool"] in {"All", "boundedCustom"}:
            assert {o["kanji"]: o["title"] for o in row["options"]} == expected_meanings
        if row["pool"] == "N1":
            assert "刀" in ids and not ids.intersection({"加", "幼"})
    assert route["actualN1CuratedAvailable"] == 1 and route["actualN1FallbackOptions"] == 2
    assert route["curatedDistractorLevels"] == {"加": 3, "幼": 2, "刀": 1}
    live = {o["kanji"]: o["title"] for o in route["exactDefaultMeaningOptions"]}
    assert live == {"松": "pine tree", "功": "merit; achievement",
                    "刀": "sword; blade", "梨": "pear tree"}
    for key in ("reciprocalLinkAndSystemBack", "meaningQuestionAndFeedback",
                "oneCuratedKeyPersistedGoodAndDone", "drawingTraceThenRecall",
                "drawingGradeRecordedOnNext", "offline"):
        assert route[key] is True, key
    assert route["touchInjectedReferenceStrokes"] == 5
    assert (route["mappedReciprocalPairs"], route["directedLinkTransitions"],
            route["activatedBaselinePairs"], route["changedReusedArticles"]) == (4, 8, 2, 3)
    assert raster["sdk"] == 35 and raster["fontScale"] == 1
    assert raster["individualGlyphProbes"] == 76 and raster["wholeWordRasterProbes"] == 8
    assert raster["panels"] == 3 and len(raster["rows"]) == 84
    assert all(r.get("hasGlyph", True) for r in raster["rows"])
    assert all(r.get("renderedCharactersNonblank", 2) == 2 for r in raster["rows"])
    assert len(list(RAW.glob("*.png"))) == 34 and len(list(SHEETS.glob("sheet-*.png"))) == 8
    return attempt, route
