def load(p):
    return json.loads(p.read_text(encoding="utf-8-sig"))


def save(p, value):
    p.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def now():
    return datetime.now(timezone.utc).isoformat()


def git(*args, repo=R):
    return subprocess.check_output(["git", "--no-pager", "-C", str(repo), *args],
                                   encoding="utf-8").strip()


def absent(pid):
    return subprocess.run(["pwsh", "-NoProfile", "-Command",
        f"if (Get-Process -Id {pid} -ErrorAction SilentlyContinue) {{ exit 1 }}"],
        capture_output=True).returncode == 0


def frozen():
    for repo, head in [(R, HEAD), (A, APP)]:
        assert git("rev-parse", "HEAD", repo=repo) == head
        assert git("log", "-1", "--format=%G?", repo=repo) == "G"
        assert not git("status", "--porcelain", "--untracked-files=all", repo=repo)
    assert git("rev-parse", "origin/main") == MAIN
    assert not git("diff", TEACHING, HEAD, "--", "kanji", "words", "articles",
                   "strokes", "manifest.json", "tools")
    p, runner = load(C / "progress.json"), load(C / "runner.json")
    assert (p["checkpoint"], p["batch"], p["status"]) == (803, "104", "continue")
    assert (runner["pid"], runner["sessionId"], runner["status"]) == (13320, L.name, "running")
    assert not absent(13320)
    assert not load(L / "preflight-804.json")["lockTouched"]

def run(label):
    frozen()
    family = label.split("-")[0]
    subprocess.run(["pwsh", "-NoProfile", "-File", str(C / "preflight-804.ps1"),
                    "-ReceiptName", f"preflight-804-before-{label}.json"], check=True)
    if label == "gpt":
        prior = load(W / "claude-execution.json")
        assert prior["exitCode"] == 0 and prior["processExited"] and absent(prior["launcherPid"])
        assert (W / "claude-preserved.json").exists(), "Archive complete first report before next launch"
        for repair in ["claude-repair03", "claude-repair04"]:
            fixed = load(W / f"{repair}-execution.json")
            assert fixed["exitCode"] == 0 and fixed["processExited"] and absent(fixed["launcherPid"])
            assert len(fixed["toolCalls"]) == 1 and fixed["allAssignedTextReturned"]
            assert fixed["multiToolResponseCount"] == 0 and not fixed["observedToolOverlaps"]
            assert (W / f"{repair}-preserved.json").exists()
    for row in load(W / "resources.json") + load(W / "image-scope.json")["images"]:
        assert sha(Path(row["path"])) == row["sha256"]
    raw = W / f"{label}.jsonl"
    assert not raw.exists(), "Never overwrite original execution"
    cmd = [shutil.which("copilot"), "-C", str(R), "--add-dir", str(C),
           "--model", MODELS[family], "--session-id", SESSIONS[family], "--no-ask-user",
           "--allow-tool=view", "--disable-builtin-mcps", "--available-tools=view",
           "--no-custom-instructions", "--output-format", "json",
           "--log-dir", str(W / f"{label}-logs"), "--usage-output-file", str(W / f"{label}-usage.json"),
           "-p", (W / f"{label}-task.txt").read_text(encoding="utf-8")]
    env = os.environ.copy()
    env.pop("GH_TOKEN", None)
    env.pop("GITHUB_TOKEN", None)
    started = now()
    with raw.open("xb") as out, (W / f"{label}-stderr.txt").open("xb") as err:
        process = subprocess.Popen(cmd, stdout=out, stderr=err, env=env)
        save(W / f"{label}-active.json", {"pid": process.pid, "sessionId": SESSIONS[family],
                                         "requestedModel": MODELS[family], "startedAt": started})
        print(f"Original-session {label} reviewer PID {process.pid}; {raw}", flush=True)
        code = process.wait()
    save(W / f"{label}-process.json", {"launcherPid": process.pid, "startedAt": started,
         "completedAt": now(), "exitCode": code, "command": cmd, "processExited": absent(process.pid)})
    audit(label)
    frozen()


def audit(label):
    events = [json.loads(line) for line in (W / f"{label}.jsonl").read_text(
        encoding="utf-8-sig").splitlines() if line.strip()]
    messages = [event["data"] for event in events if event.get("type") == "assistant.message"]
    report = W / f"{label}.md"
    assert not report.exists()
    report.write_text((messages[-1].get("content", "") if messages else "No completed report.") + "\n",
                      encoding="utf-8")
    resources = {row["path"]: row for row in load(W / "resources.json")}
    images = {row["path"]: row for row in load(W / "image-scope.json")["images"]}
    scope_path = W / f"{label}-scope.json"
    if scope_path.exists():
        scope = load(scope_path)
        resources = {path: row for path, row in resources.items() if path in scope["resources"]}
        images = {path: row for path, row in images.items() if path in scope["images"]}
    returned = {path: set() for path in resources}
    calls, active, overlaps, image_opens = {}, set(), [], []
    for event in events:
        data = event.get("data", {})
        if event.get("type") == "tool.execution_start":
            key = data["toolCallId"]
            if active:
                overlaps.append({"starting": key, "active": sorted(active)})
            active.add(key)
            calls[key] = {"name": data["toolName"], "arguments": data["arguments"],
                          "startedAt": event["timestamp"]}
        elif event.get("type") == "tool.execution_complete":
            key = data["toolCallId"]
            active.discard(key)
            result = data.get("result", {}).get("content", "")
            if not isinstance(result, str):
                result = json.dumps(result, ensure_ascii=False)
            call = calls[key]
            call.update(completedAt=event["timestamp"], success=data["success"],
                        returnedCharacters=len(result), returnedSha256=hashlib.sha256(result.encode()).hexdigest())
            path = call["arguments"].get("path")
            if path in resources:
                lines = Path(path).read_text(encoding="utf-8").splitlines(keepends=True)
                a, b = call["arguments"].get("view_range", [1, -1])
                expected = "".join(lines[a - 1:None if b == -1 else b])
                norm = lambda text: text.replace("\r\n", "\n").rstrip()
                exact = norm(expected) in norm(result)
                call["exactRequestedTextContainedInResult"] = exact
                if exact and data["success"]:
                    returned[path].update(range(a, len(lines) + 1 if b == -1 else b + 1))
            if path in images:
                payloads = []
                original = Path(path).read_bytes()
                for item in data.get("result", {}).get("binaryResultsForLlm", []):
                    if item.get("type") == "image" and isinstance(item.get("data"), str):
                        content = base64.b64decode(item["data"], validate=True)
                        payloads.append({"sha256": hashlib.sha256(content).hexdigest(),
                            "bytes": len(content), "equalsOriginalBytes": content == original,
                            "pngDimensions": list(struct.unpack(">II", content[16:24])),
                            "originalPngDimensions": list(struct.unpack(">II", original[16:24]))})
                call["imagePayloads"] = payloads
                if data["success"]:
                    image_opens.append({"path": path, "payloads": payloads})
    coverage = [{"path": path, "lines": row["lines"],
                 "missingLines": sorted(set(range(1, row["lines"] + 1)) - returned[path])}
                for path, row in resources.items()]
    usage = load(W / f"{label}-usage.json") if (W / f"{label}-usage.json").exists() else None
    family = label.split("-")[0]
    record = {
        **load(W / f"{label}-process.json"), "snapshot": HEAD, "unchangedTeachingSnapshot": TEACHING,
        "sessionId": SESSIONS[family], "requestedModel": MODELS[family],
        "actualModels": sorted({message.get("model", "") for message in messages}),
        "usageCurrentModel": usage.get("currentModel") if usage else None,
        "availableTools": ["view"], "toolCalls": list(calls.values()),
        "configurationEvidence": [event["data"] for event in events if event.get("type") in [
            "session.info", "session.tools_updated"]],
        "warnings": [event for event in events if event.get("type") == "session.warning"],
        "compactions": [event for event in events if event.get("type") == "session.compaction_complete"],
        "observedToolOverlaps": overlaps, "unfinishedTools": sorted(active),
        "multiToolResponseCount": sum(len(message.get("toolRequests", [])) > 1 for message in messages),
        "textCoverage": coverage, "allAssignedTextReturned": not any(row["missingLines"] for row in coverage),
        "successfulImageOpens": image_opens,
        "missingImages": sorted(set(images) - {row["path"] for row in image_opens}),
        "rawExecutionSha256": sha(W / f"{label}.jsonl"), "reportSha256": sha(report), "usage": usage,
        "limits": "Actual source/image/tool/process delivery, not comprehension, image retention in every model request, original-pixel/font-face/phone certification.",
    }
    assert not (W / f"{label}-execution.json").exists()
    save(W / f"{label}-execution.json", record)
    print(json.dumps({key: record[key] for key in ["actualModels", "exitCode", "processExited",
        "allAssignedTextReturned", "observedToolOverlaps", "multiToolResponseCount", "missingImages"]}))


def preserve(label):
    record = load(W / f"{label}-execution.json")
    family = label.split("-")[0]
    assert record["exitCode"] == 0 and record["processExited"] and absent(record["launcherPid"])
    assert record["actualModels"] == [MODELS[family]] and record["usageCurrentModel"] == MODELS[family]
    archive = W / f"{label}.jsonl.gz"
    assert not archive.exists()
    raw = (W / f"{label}.jsonl").read_bytes()
    archive.write_bytes(gzip.compress(raw, compresslevel=9, mtime=0))
    assert gzip.decompress(archive.read_bytes()) == raw
    save(W / f"{label}-preserved.json", {"completeReport": str(W / f"{label}.md"),
        "reportSha256": sha(W / f"{label}.md"), "rawExecutionSha256": hashlib.sha256(raw).hexdigest(),
        "losslessArchive": str(archive), "archiveSha256": sha(archive), "processExited": True,
        "recordedAt": now(), "limits": "Preservation only; assess report and actual coverage separately."})
    print(f"Complete {label} report and exact execution losslessly preserved; process exited.")
