def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def verify_runtime():
    sys.path.insert(0, str(R / "tools"))
    from build_manifest import build_manifest, sha256
    manifest = m.load(R / "manifest.json")
    assert manifest == build_manifest(R)
    assert manifest["contentVersion"] == "a5bc9b199c7df334"
    count = 0
    for section, entries in manifest.items():
        if not isinstance(entries, dict):
            continue
        for name, expected in entries.items():
            path = R / Path(*name.split("/"))
            actual = m.sha(path) if section in {
                "images", "audio", "listeningTaskAudio", "listeningTaskImages"} else sha256(path)
            assert actual == expected, name
            count += 1
    assert count == 8466
    assert not m.git("diff", m.HEAD, "--", "kanji", "words", "articles",
                     "strokes", "manifest.json", "tools")
    return count

    payload_directory = D / "image-payloads"
    payload_directory.mkdir()
    archives = []
    for source in sorted(W.iterdir()):
        if source.is_dir() or source.suffix == ".gz":
            continue
        raw = source.read_bytes()
        if source.suffix != ".jsonl":
            target = D / (source.name + ".gz")
            target.write_bytes(gzip.compress(raw, compresslevel=9, mtime=0))
            assert gzip.decompress(target.read_bytes()) == raw
            archives.append({"archive": target.name, "archiveSha256": m.sha(target),
                             "rawSha256": digest(raw), "rawBytes": len(raw)})
            continue
        transformed, replacements = [], {}
        for line in raw.splitlines(keepends=True):
            event = json.loads(line)
            if event["type"] == "tool.execution_complete":
                for item in event.get("data", {}).get("result", {}).get("binaryResultsForLlm", []):
                    if item.get("type") == "image" and isinstance(item.get("data"), str):
                        encoded = item["data"].encode("ascii")
                        image = base64.b64decode(encoded, validate=True)
                        assert base64.b64encode(image) == encoded
                        image_hash = digest(image)
                        marker = ("__N1_IMAGE_PAYLOAD_SHA256_" + image_hash + "__").encode("ascii")
                        assert marker not in raw and encoded in line
                        target = payload_directory / (image_hash + ".png")
                        if target.exists():
                            assert target.read_bytes() == image
                        else:
                            target.write_bytes(image)
                        replacements[marker] = encoded
                        line = line.replace(encoded, marker)
            transformed.append(line)
        factored = b"".join(transformed)
        target = D / (source.name + ".factored.gz")
        target.write_bytes(gzip.compress(factored, compresslevel=9, mtime=0))
        restored = gzip.decompress(target.read_bytes())
        for marker, encoded in replacements.items():
            restored = restored.replace(marker, encoded)
        assert restored == raw
        archives.append({"archive": target.name, "archiveSha256": m.sha(target),
                         "factoredSha256": digest(factored), "rawSha256": digest(raw),
                         "rawBytes": len(raw), "distinctPayloadsUsed": len(replacements),
                         "payloadMarker": "__N1_IMAGE_PAYLOAD_SHA256_<sha256>__",
                         "reconstruction": "Decompress; replace markers with canonical base64 of named PNG. EVERY original stdout byte/hash round-tripped exactly, including all events and tool results."})

def seal():
    assert m.git("rev-parse", "HEAD") == m.HEAD
    for row in m.load(D / "archive-integrity.json"):
        packed = D / row["archive"]
        assert m.sha(packed) == row["archiveSha256"]
        restored = gzip.decompress(packed.read_bytes())
        if "factoredSha256" in row:
            assert digest(restored) == row["factoredSha256"]
            for payload in m.load(D / "image-payload-index.json"):
                image = (D / payload["file"]).read_bytes()
                assert digest(image) == payload["sha256"]
                marker = ("__N1_IMAGE_PAYLOAD_SHA256_" + payload["sha256"] + "__").encode("ascii")
                restored = restored.replace(marker, base64.b64encode(image))
        assert len(restored) == row["rawBytes"] and digest(restored) == row["rawSha256"]
    count = verify_runtime()
    validation = L / "validate-804.txt"
    assert validation.read_text(encoding="utf-8-sig").rstrip().endswith("0 error(s)")
    target = D / "validate-804.txt.gz"
    assert not target.exists()
    target.write_bytes(gzip.compress(validation.read_bytes(), mtime=0))
    assert gzip.decompress(target.read_bytes()) == validation.read_bytes()
    e = m.load(E)
    e["validation"] = {
        "commands": ["python -B tools\\validate.py", "python -B tools\\build_manifest.py",
                     "git diff --exit-code -- manifest.json", "git diff --check"],
        "observedTerminalExitCode": 0, "validatorErrors": 0, "manifestUnchanged": True,
        "runtimeHashesVerified": count, "contentVersion": "a5bc9b199c7df334",
