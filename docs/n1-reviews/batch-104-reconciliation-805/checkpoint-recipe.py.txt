def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def save(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def git(*args, repo=R):
    return subprocess.check_output(
        ["git", "--no-pager", "-C", str(repo), *args], encoding="utf-8").strip()


def now():
    return datetime.now(timezone.utc).isoformat()


def replace_progress(value, suffix):
    tmp = C / f"progress.json.805-{suffix}.tmp"
    assert not tmp.exists()
    save(tmp, value)
    assert load(tmp) == value
    os.replace(tmp, C / "progress.json")
    assert load(C / "progress.json") == value

def runtime():
    sys.path.insert(0, str(R / "tools"))
    from build_manifest import build_manifest, sha256
    manifest = load(R / "manifest.json")
    assert manifest == build_manifest(R)
    assert manifest["contentVersion"] == VERSION
    count = 0
    for section, entries in manifest.items():
        if not isinstance(entries, dict):
            continue
        for name, expected in entries.items():
            path = R / Path(*name.split("/"))
            actual = sha(path.read_bytes()) if section in {
                "images", "audio", "listeningTaskAudio", "listeningTaskImages"} else sha256(path)
            assert actual == expected, name
            count += 1
    assert count == 8466
    assert not git("diff", HEAD, "--", "kanji", "words", "articles", "strokes", "manifest.json", "tools")
    return count
