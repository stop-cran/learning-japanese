def load(p):
    return json.loads(p.read_text(encoding="utf-8-sig"))


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def save(p, data):
    assert not p.exists(), p
    p.write_text(json.dumps(data, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(p) == data


def git(*args, root=R):
    return subprocess.check_output(["git", "--no-pager", "-C", str(root), *args], encoding="utf-8").strip()

def runtime():
    m = load(R / "manifest.json")
    assert m == json.loads(git("show", f"{BASE}:manifest.json"))
    hashes = {}
    for section, entries in m.items():
        if not isinstance(entries, dict):
            continue
        for name, digest in entries.items():
            assert name not in hashes
            data = (R / name).read_bytes()
            if section not in {"images", "audio", "listeningTaskAudio", "listeningTaskImages"}:
                data = data.replace(b"\r\n", b"\n")
            assert hashlib.sha256(data).hexdigest() == digest, name
            hashes[name] = digest
    assert len(hashes) == 8469 and m["contentVersion"] == "5572c6c467e6b73b"
    return {"contentVersion": m["contentVersion"], "runtimeHashesVerified": len(hashes),
            "manifestSha256": sha(R / "manifest.json")}

def plain(text):
    return re.sub(r"\[([^\]]+)\]\([^)]+\)", r"\1", text).replace("**", "").replace("`", "")


def ks(text):
    return json.dumps(text, ensure_ascii=False)


def kl(values):
    return "listOf(" + ", ".join(map(ks, values)) + ")"


def body(name):
    return (R / name).read_text(encoding="utf-8").split("---", 2)[2].strip()


def blocks(name):
    return body(name).split("\n\n")


def heading(name):
    return plain(body(name).splitlines()[0][2:])


def sections(name, prefix):
    rows = []
    for block in blocks(name):
        if block.startswith("# "):
            continue
        if block.startswith("## "):
            rows.append(f"            section({ks(block[3:].strip())})")
            continue
        for line in (block.splitlines() if block.startswith("|") or block.startswith("- ") else [block]):
            if line.startswith("|"):
                cells = [cell.strip() for cell in line.strip("|").split("|")]
                if cells[0] == "Reading" or cells[0].startswith("---"):
                    continue
                for cell in (cells[0], cells[2]):
                    rows.append(f'            show(readingExample({ks(plain(cell))}), "{prefix}-{len(rows):02d}")')
            else:
                text = plain(" ".join(line.removeprefix("- ").splitlines()))
                rows.append(f'            show(articleBullet({ks(text)}), "{prefix}-{len(rows):02d}")')
    return "\n".join(rows)
