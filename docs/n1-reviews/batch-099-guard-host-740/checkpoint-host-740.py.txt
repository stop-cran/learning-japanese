def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(path, value):
    assert not path.exists(), path
    path.write_text(json.dumps(value, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(path) == value


def git(*args, root=R):
    return subprocess.check_output(
        ["git", "--no-pager", "-C", str(root), *args], encoding="utf-8"
    ).strip()


def context():
    p = load(C / "progress.json")
    assert (p["checkpoint"], p["status"], p["batch"]) == (739, "continue", "099")
    runner = load(C / "runner.json")
    assert (runner["status"], runner["sessionId"], runner["pid"]) == ("running", SESSION, 5920)
    assert git("rev-parse", "HEAD", root=A) == APP
    assert git("log", "-1", "--format=%G?", root=A) == "G"
    assert not git("status", "--porcelain", "--untracked-files=all", root=A)
    return p


def replace_progress(current, after, phase):
    temporary = C / f"progress.json.740-{phase}.tmp"
    save(temporary, after)
    assert load(C / "progress.json") == current
    os.replace(temporary, C / "progress.json")
    assert load(C / "progress.json") == after

    D.mkdir(exist_ok=True)
    reports = []
    for module, task, name, expected in specs:
        source = A / module / "build" / "test-results" / task / f"TEST-{name}.xml"
        tree = ET.parse(source).getroot()
        assert int(tree.attrib["tests"]) == len(expected)
        assert all(int(tree.attrib[k]) == 0 for k in ("failures", "errors", "skipped"))
        cases = tree.findall("testcase")
        assert {case.attrib["name"] for case in cases} == expected
        assert all(not case.findall("skipped") and not case.findall("failure") and not case.findall("error") for case in cases)
        timestamp = datetime.fromisoformat(tree.attrib["timestamp"].replace("Z", "+00:00"))
        if timestamp.tzinfo is None:
            timestamp = timestamp.replace(tzinfo=timezone.utc)
        assert timestamp >= start_time
        target = D / source.name
        if target.exists():
            assert sha(target) == sha(source)
        else:
            target.write_bytes(source.read_bytes())
        assert sha(target) == sha(source)
        reports.append({
            "name": name, "tests": len(expected), "failures": 0, "errors": 0, "skipped": 0,
            "timestamp": tree.attrib["timestamp"], "caseNames": sorted(expected),

        })
    manifest = load(R / "manifest.json")
    preparation = load(R / "docs" / "n1-reviews" / "batch-099-guard-native-739" / "preparation.json")
    archive_path = Path(preparation["archive"])
    assert sha(archive_path) == preparation["archiveSha256"]
    with zipfile.ZipFile(archive_path) as z:
        manifests = [n for n in z.namelist() if n.endswith("/manifest.json") or n == "manifest.json"]
        assert len(manifests) == 1
        archive_manifest = json.loads(z.read(manifests[0]))
        assert archive_manifest == manifest
        prefix = manifests[0][:-len("manifest.json")]
        entries = {}
        for section_name, section in manifest.items():
            if isinstance(section, dict):
                for name, digest in section.items():
                    assert name not in entries
                    entries[name] = (digest, section_name in {
                        "images", "audio", "listeningTaskAudio", "listeningTaskImages",
                    })
        for name, (digest, binary) in entries.items():
            local = (R / name).read_bytes()
            packaged = z.read(prefix + name)
            if not binary:
                local = local.replace(b"\r\n", b"\n")
                packaged = packaged.replace(b"\r\n", b"\n")
            assert local == packaged, name
            assert hashlib.sha256(local).hexdigest() == digest, name
            assert hashlib.sha256(packaged).hexdigest() == digest, name
        assert len(entries) == 8451
