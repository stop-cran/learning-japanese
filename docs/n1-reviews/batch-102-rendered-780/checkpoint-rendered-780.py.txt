    assert not D.exists()
    D.mkdir()
    index = []

    def store(source, name=None, compress=False):
        source = Path(source)
        name = source.name if name is None else name
        data = source.read_bytes()
        destination = D / (name + ".gz" if compress else name)
        assert not destination.exists()
        destination.write_bytes(gzip.compress(data, mtime=0) if compress else data)
        if compress:
            assert gzip.decompress(destination.read_bytes()) == data
        index.append({"file": destination.name, "sourcePath": str(source),
                      "sourceBytes": len(data), "sourceSha256": r.sha(source),
                      "storedSha256": r.sha(destination), "losslessGzip": compress})

    for path in sorted(W.iterdir()):
        if path.is_file():
            compress = path.suffix in {".txt", ".xml", ".jsonl"} or path.stat().st_size > 100000
            store(path, compress=compress)
