def load(p):
    return json.loads(p.read_text(encoding="utf-8-sig"))


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def save(p, data, replace=False):
    assert replace or not p.exists(), p
    temp = p.with_suffix(p.suffix + ".795.tmp")
    assert not temp.exists(), temp
    temp.write_text(json.dumps(data, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(temp) == data
    os.replace(temp, p)
    assert load(p) == data


def output(args, root=R):
    return subprocess.check_output(args, cwd=root, encoding="utf-8").strip()


def git(*args, root=R):
    return output(["git", "--no-pager", "-C", str(root), *args])


def context(head, label):
    assert not os.environ.get("GH_TOKEN") and not os.environ.get("GITHUB_TOKEN")
    name = f"preflight-795-{label}.json"
    subprocess.run(["pwsh", "-NoProfile", "-NonInteractive", "-File",
                    str(C / "preflight-795.ps1"), "-ExpectedContentHead", head,
                    "-ReceiptName", name], check=True)
    return load(L / name)


def runtime():
    from build_manifest import build_manifest
    manifest = load(R / "manifest.json")
    assert manifest == build_manifest(R) and manifest["contentVersion"] == VERSION
    count = 0
    for section, entries in manifest.items():
        if not isinstance(entries, dict):
            continue
        for name, value in entries.items():
            raw = (R / Path(*name.split("/"))).read_bytes()
            binary = section in {"images", "audio", "listeningTaskAudio", "listeningTaskImages"}
            assert digest(raw if binary else raw.replace(b"\r\n", b"\n")) == value, name
            count += 1
    assert count == 8463
    assert not git("diff", TEACHING, "--", "manifest.json", "kanji", "words", "strokes",
                   "articles", "tools")
    return dict(contentVersion=VERSION, runtimeHashesVerified=count,
                manifestSha256=digest((R / "manifest.json").read_bytes()),
                unchangedSinceTeaching=TEACHING)

def canonical(e):
    return e.tag, tuple(sorted(e.attrib.items())), (e.text or "").strip(), tuple(canonical(c) for c in e)


def source_audit():
    from audit_n1_batch import audit
    from audit_n1_pilot import metadata

    from generate_strokes import convert, KANJIVG_REVISION
    from fontTools.ttLib import TTFont

    ledger, plan = load(S / "batch-103-progress.json"), load(S / "batch-103-authoring-plan.json")
    snapshot = load(R / "docs" / "n1-source-snapshot.json")
    inputs = {}
    for name, row in snapshot["inputs"].items():
        raw = (C / "inputs" / name).read_bytes()
        assert len(raw) == row["bytes"] and digest(raw) == row["sha256"]
        inputs[name] = raw
    jm = inputs["n1-implementation-JMdict_e.gz"]
    kd = gzip.decompress(inputs["n1-planning-kanjidic2.xml.gz"])
    assert digest(kd) == snapshot["kanjidic2"]["xmlSha256"]
    assert digest(gzip.decompress(jm)) == snapshot["jmdict"]["xmlSha256"]
    packets = []
    for row in ledger["dictionaryPackets"]:
        ids = load(S / row["selection"])["entryIds"]
        raw = extract(jm, snapshot["inputs"]["n1-implementation-JMdict_e.gz"]["sha256"], ids)
        assert raw == (S / row["xml"]).read_bytes()
        packets.append(dict(xml=row["xml"], completeFamilies=len(ids), sha256=digest(raw)))
    records = {e.findtext("literal"): canonical(e) for e in ET.parse(
        S / "batch-103-comparison-kanjidic2.xml").getroot().findall("character")}
    actual = {}
    for _, e in ET.iterparse(io.BytesIO(kd), events=["end"]):
        if e.tag == "character":
            char = e.findtext("literal")
            if char in records:
                actual[char] = canonical(e)
            e.clear()
    assert records == actual and len(records) == 11
    assert canonical(ET.parse(S / "batch-103-kanjidic2.xml").getroot().find("character")) == actual[MEMBER]
    pages = load(S / "batch-103-public-source-receipts.json")
    for row in pages:
        raw = (S / row["archive"]).read_bytes()
        assert digest(raw) == row["archiveSha256"]
        assert digest(gzip.decompress(raw)) == row["rawSha256"]
        assert digest((S / row["extractedText"]).read_bytes()) == row["extractedTextSha256"]
    counts = dict(kanji=0, words=0, strokes=0)
    for name, row in load(S / "baseline-inventory.json")["files"].items():
        kind = name.split("/")[0]
        if kind in {"kanji", "words"}:
            current = metadata(R, name)
            for key, value in row["metadata"].items():
                if key == "quiz_exclusions":
                    assert set(value) <= set(current.get(key, [])), name
                elif name == "kanji/\u5c71.md" and key == "distractors":
                    assert current[key] == list("\u5ddd\u51fa\u6728")
                else:
                    assert current[key] == value, (name, key)
            counts[kind] += 1
        elif kind == "strokes":
            assert digest((R / name).read_bytes()) == row["sha256"], name
            counts[kind] += 1
    assert counts == dict(kanji=985, words=4942, strokes=985)
    named = load(C / "inputs" / "n1-kanji-source-20261102.json")
    assert named == load(S / "community-list-20261102.json")
    assert ledger["sourceRows"] == [114] and named[113] == MEMBER
    chars = {p.stem for p in (R / "kanji").glob("*.md")}
    old = json.loads(git("show", f"{MAIN}:manifest.json"))
    old_chars = {Path(p).stem for p in old["files"] if p.startswith("kanji/")}
    assert chars - old_chars == {MEMBER} and not old_chars - chars
    assert metadata(R, "kanji/\u5996.md")["jlpt"] == 1
    assert all(metadata(R, f"kanji/{c}.md")["jlpt"] == 2 for c in "\u4fdd\u6848\u6d3e\u7b2c\u7b56\u7d50")
    assert metadata(R, f"kanji/{MEMBER}.md") == plan["proposedMetadata"]
    for row in plan["newSupportMetadata"]:
        assert metadata(R, f"words/{row['word']}.md") == row and "jlpt" not in row
    pairs = {tuple(p) for p in plan["allMappedReciprocalPairs"]}
    assert pairs == {(r["kanji"], r["article"]) for r in ledger["supportMappings"]}
    assert len(pairs) == 3
    for char, name in pairs:
        assert f"(../kanji/{char}.md)" in (R / name).read_text(encoding="utf-8")
        assert f"(../{name})" in (R / f"kanji/{char}.md").read_text(encoding="utf-8")
    activated = {f"words/{p.name}" for p in (R / "words").glob("*.md")
                 if MEMBER in metadata(R, f"words/{p.name}")["kanji"]}
    assert activated == {name for _, name in pairs}
    for row in load(S / "batch-103-comparison-selection.json")["articleSnapshots"]:
        assert metadata(R, row["article"]) == row["metadata"]
        if row["article"] not in {r["article"] for r in ledger["changedReusedPassages"]}:
            assert digest((R / row["article"]).read_bytes()) == row["sha256"]
    guards = {tuple(sorted([row["word"], other]))
              for row in plan["newSupportMetadata"] for other in row["quiz_exclusions"]}
    assert len(guards) == 27
    assert plan["proposedMetadata"]["quiz_exclusions"] == list("\u52aa\u52c9\u52e4\u652f")
    svg, proof = (S / "batch-103-kanjivg.svg").read_bytes(), load(S / "batch-103-stroke-evidence.json")
    assert digest(svg) == proof["sha256"]
    assert hashlib.sha1(f"blob {len(svg)}\0".encode() + svg).hexdigest() == proof["gitBlob"]
    assert proof["revision"] == KANJIVG_REVISION
    assert digest((R / "tools" / "generate_strokes.py").read_bytes().replace(b"\r\n", b"\n")) == proof["converterNormalizedSha256"]
    geometry = load(R / "strokes" / f"{MEMBER}.json")

        assert path.is_relative_to(A) and digest(path.read_bytes()) == row["sha256"]
        with TTFont(path) as font:

        assert missing == row["missing"] == ["\u5382"]
        fonts.append(dict(path=str(path), sha256=row["sha256"], missing=missing))
    current_counts = dict(kanji=len(chars), words=len(list((R / "words").glob("*.md"))),
        n1Labels=sum(metadata(R, f"kanji/{c}.md")["jlpt"] == 1 for c in chars),
        namedSourceMembersPresent=len(chars & set(named)))
    assert current_counts == dict(kanji=1126, words=5077, n1Labels=135, namedSourceMembersPresent=140)

        completeRawCharacterRecordsVerified=len(records), archivedPublicPagesVerified=len(pages),
        originalBaselineMetadataAndStrokesPreserved=counts, exactNewMember=MEMBER,

        pinnedGeometryExactlyRegenerated=True, generatedStrokes=7,
