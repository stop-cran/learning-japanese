def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(path, value):

    path.write_text(json.dumps(value, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")

def plain(text):
    return re.sub(r"\[([^\]]+)\]\([^)]+\)", r"\1", text).replace("**", "").replace("`", "")


def ks(text):
    return json.dumps(text, ensure_ascii=False)


def kl(values):
    return "listOf(" + ", ".join(map(ks, values)) + ")"


def body(name):
    return (R / name).read_text(encoding="utf-8").split("---", 2)[2].strip()


def heading(name):
    return plain(body(name).splitlines()[0][2:])


def blocks(name):
    return body(name).split("\n\n")


def show_sections(name, prefix):
    rows = []
    for block in blocks(name):
        if block.startswith("# "):
            continue
        if block.startswith("## "):
            rows.append(f"            section({ks(block[3:].strip())})")
            continue
        for line in (block.splitlines() if block.startswith("|") or block.startswith("- ") else [block]):
            if line.startswith("|"):
                cells = [cell.strip() for cell in line.strip("|").split("|")]
                if cells[0] == "Reading" or cells[0].startswith("---"):
                    continue
                for cell in (cells[0], cells[2]):
                    rows.append(f"            show(readingExample({ks(plain(cell))}), \"{prefix}-{len(rows):02d}\")")
            else:
                text = plain(" ".join(line.removeprefix("- ").splitlines()))
                rows.append(f"            show(articleBullet({ks(text)}), \"{prefix}-{len(rows):02d}\")")
    return "\n".join(rows)
