def load(p):
    return json.loads(p.read_text(encoding="utf-8-sig"))


def save(p, value):
    assert not p.exists(), p
    p.write_text(json.dumps(value, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(p) == value


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def git(*args, root=R):
    return subprocess.check_output(
        ["git", "--no-pager", "-C", str(root), *args], encoding="utf-8"
    ).strip()


def runtime():
    e = load(R / "docs" / "n1-sources" / "execution-752.json")
    archive = Path(e["preparation"]["archive"])
    assert sha(archive) == e["preparation"]["archiveSha256"]
    with zipfile.ZipFile(archive) as z:
        manifest = load(R / "manifest.json")
        keys = [n for n in z.namelist() if n.endswith("/manifest.json") or n == "manifest.json"]
        assert len(keys) == 1
        key = keys[0]
        assert json.loads(z.read(key)) == manifest
        prefix = key[:-len("manifest.json")]
        count = 0
        for section_name, section in manifest.items():
            if not isinstance(section, dict):
                continue
            for name, expected in section.items():
                raw = (R / name).read_bytes()
                packaged = z.read(prefix + name)
                if section_name not in {"images", "audio", "listeningTaskAudio", "listeningTaskImages"}:
                    raw = raw.replace(b"\r\n", b"\n")
                    packaged = packaged.replace(b"\r\n", b"\n")
                assert hashlib.sha256(raw).hexdigest() == expected, name
                assert raw == packaged, name
                count += 1
        assert count == 8453, count
        assert manifest["contentVersion"] == "82b45827a5bb4ae3"
    return count


def replace_progress(before, after, phase):

    assert load(C / "progress.json") == after
