def verify_evidence():
    evidence = load(E)
    for entry in evidence["provenance"]:
        assert sha(R / entry["archive"]) == entry["sha256"]
        if "originalBytesArchive" in entry:
            assert sha(R / entry["originalBytesArchive"]) == entry["originalBytesArchiveSha256"]
            assert H.hashlib.sha256(gzip.decompress((R / entry["originalBytesArchive"]).read_bytes())).hexdigest() == entry["sourceSha256"]
    prep = evidence["preparation"]
    assert sha(FIXTURE) == prep["fixtureSha256"] and sha(ARCHIVE) == prep["archiveSha256"]
    assert sha(W / "fixture-from-preserved812.diff.gz") == prep["fixtureDiffSha256"]
    for name in ("app", "test"):
        assert sha(Path(evidence["compilation"][name + "Apk"])) == evidence["compilation"][name + "ApkSha256"]
    assert H.runtime()["manifestSha256"] == prep["manifestSha256"]
    assert evidence["compilation"]["kotlinFixtureCompilationExecuted"]
    assert not evidence["nativeExecuted"] and not evidence["runtimePublished"]
    assert not git("status", "--porcelain", "--untracked-files=all", root=A)
    return evidence
