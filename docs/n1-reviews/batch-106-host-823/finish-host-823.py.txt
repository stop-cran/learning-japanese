def load(p):
    return json.loads(p.read_text(encoding="utf-8-sig"))


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def git(*args, root=R):
    return subprocess.check_output(["git", "--no-pager", "-C", str(root), *args], encoding="utf-8").strip()


def write(p, data, *, new=False):
    if new:
        assert not p.exists(), p
    p.write_text(json.dumps(data, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(p) == data


def context():
    p = load(C / "progress.json")
    assert (p["checkpoint"], p["status"], p["batch"]) == (822, "continue", "106")
    runner = load(C / "runner.json")
    assert (runner["status"], runner["pid"], runner["sessionId"]) == ("running", 13320, L.name)
    assert git("rev-parse", "HEAD", root=A) == APP and git("log", "-1", "--format=%G?", root=A) == "G"
    assert not git("status", "--porcelain", "--untracked-files=all", root=A)
    return p


def runtime():
    m = load(R / "manifest.json")
    assert m == json.loads(git("show", f"{BASE}:manifest.json"))
    hashes = {}
    for section, entries in m.items():
        if not isinstance(entries, dict):
            continue
        for path, digest in entries.items():
            assert path not in hashes
            data = (R / path).read_bytes()
            if section not in {"images", "audio", "listeningTaskAudio", "listeningTaskImages"}:
                data = data.replace(b"\r\n", b"\n")
            assert hashlib.sha256(data).hexdigest() == digest, path
            hashes[path] = digest
    assert len(hashes) == 8472 and m["contentVersion"] == "89809c00e1676979"
    return {"contentVersion": m["contentVersion"], "runtimeHashesVerified": len(hashes),
            "manifestSha256": sha(R / "manifest.json"), "runtimeUnchanged": True}


def audit():
    context()
    assert git("rev-parse", "HEAD") == BASE and git("log", "-1", "--format=%G?") == "G"
    untracked = git("ls-files", "--others", "--exclude-standard").splitlines()
    changed = git("diff", "--name-only", BASE).splitlines() + untracked
    assert changed and all(x.startswith("docs/n1-reviews/batch-106-host-823/")
                           or x == "docs/n1-sources/execution-823.json" for x in changed)
    e, s, labels = load(E), load(D / "finite-semantic-disposition.json"), load(D / "sampled-labels.json")
    assert s["snapshot"] == BASE and s["status"] == "accepted-bounded-target-samples"
    for section, field in [("kanji", "kanjiDisposition"), ("words", "wordDisposition")]:
        ids = [i for row in s[field] for i in row["ids"]]
        assert len(ids) == len(set(ids)) and set(ids) == {x[0] for x in labels[section]}, field
    assert s["readingDepth"]["sampledKanjiLabelsIncludingTarget"] == e["sampledKanjiLabels"] == 41
    assert s["readingDepth"]["sampledWordsIncludingTarget"] == e["sampledWords"] == 22
    assert s["readingDepth"]["completeLexicalFamiliesInspectedForAllSensesAndRestrictions"] == e["completeFamilies"] == 32
    assert s["readingDepth"]["fullRawCharacterRecordsInspectedForAllEnglishMeanings"] == e["fullCharacterRecords"] == 44
    assert len(e["completeWordArticlePacket"]) == 29
    assert len((L / "finite-articles-823.txt").read_text(encoding="utf-8").splitlines()) == 1358
    assert not s["runtimeCorrectionNeeded"]
    assert s["additionalHypotheses"][0]["status"] == "unresolved-nonblocking-hypothesis-preserved"
    assert e["totalTests"] == 5 and all(e[k] == 0 for k in ("failures", "errors", "skipped"))
    assert sum(row["tests"] for row in e["tests"]) == 5
    for record in e["provenance"] + e["compressedEvidence"] + e["tests"]:
        assert sha(R / record["archive"]) == record["sha256"]
        if "rawSha256" in record:
            assert hashlib.sha256(gzip.decompress((R / record["archive"]).read_bytes())).hexdigest() == record["rawSha256"]
    assert runtime()["manifestSha256"] == e["manifestSha256"]
    print("All finite labels/dispositions, source families, fresh XML, evidence and8472runtime hashes verified.")


def package():
    audit()
    e = load(E)
    assert e["stage"] == "fresh-host-passed-finite-semantic-disposition-pending"
    for name in ["sampled-labels.json", "sampled-source-selection.json", "sampled-jmdict.xml",
                 "sampled-kanjidic2.xml", "finite-semantic-disposition.json"]:
        target = D / name
        e["provenance"].append({"source": "generated/source-bound contributor evidence",
                                "archive": target.relative_to(R).as_posix(), "sha256": sha(target)})
    for source in [Path(__file__), C / "checks-host-823.ps1", C / "post-host-823.ps1",
                   L / "preflight-823-checks.json"]:
        target = D / (source.name + (".txt" if source.suffix in {".py", ".ps1"} else ""))
        assert not target.exists()
        target.write_bytes(source.read_bytes())
        e["provenance"].append({"source": str(source), "archive": target.relative_to(R).as_posix(), "sha256": sha(target)})
    for name in ["validate-823.txt", "manifest-823.txt", "manifest-diff-823.txt",
                 "budget-823.txt", "vocabulary-823.txt", "host-audit-823.txt",
                 "host-audit-823-retry.txt", "whitespace-823.txt"]:
        source, target = L / name, D / (name + ".gz")
        assert not target.exists()
        target.write_bytes(gzip.compress(source.read_bytes(), mtime=0))
        e["compressedEvidence"].append({"source": str(source), "archive": target.relative_to(R).as_posix(),
                                       "rawSha256": sha(source), "sha256": sha(target)})
    for record in e["provenance"] + e["tests"]:
        path = R / record["archive"]
        raw = path.read_bytes()
        assert hashlib.sha256(raw).hexdigest() == record["sha256"]
        data = raw.replace(b"\r\n", b"\n")
        if raw == data:
            continue
        raw_path = path.with_name(path.name + ".original-bytes.gz")
        assert not raw_path.exists()
        raw_path.write_bytes(gzip.compress(raw, mtime=0))
        path.write_bytes(data)
        record.update(sourceRawSha256=hashlib.sha256(raw).hexdigest(),
                      sourceRawArchive=raw_path.relative_to(R).as_posix(),
                      sha256=hashlib.sha256(data).hexdigest())
        e["compressedEvidence"].append({"source": "exact original captured bytes before text normalization",
            "archive": raw_path.relative_to(R).as_posix(),
            "rawSha256": hashlib.sha256(raw).hexdigest(), "sha256": sha(raw_path)})
    e["archiveByteConventions"] = "Text hashes name LF-normalized committed bytes. Any differing original CRLF capture is preserved byte-exact in original-bytes.gz; compressed evidence remains exact. No Git attribute/configuration change."
    e["finiteSemanticAcceptance"] = {
        "status": "accepted-bounded-target-samples", "role": "author/coordinator, not independent review",
        "snapshot": BASE, "disposition": (D / "finite-semantic-disposition.json").relative_to(R).as_posix(),
        "sampledKanjiLabelsIncludingTarget": 41, "sampledWordsIncludingTarget": 22,
        "weakKunHypothesis": "Original822 weak勅/訓hypothesis preserved unresolved and nonblocking. Explicit32pair samples prove availability only; no new guard/forced withdrawal/universal safety claim.",
        "runtimeCorrectionNeeded": False, "limits": load(D / "finite-semantic-disposition.json")["limits"]}
    e.update(stage="batch-106-host-passed-awaiting-native-preparation", nextAction=NEXT)
    e["auditDiagnostic"] = {
        "originalFailure": "The evidence-only article-line assertion used the view tool's1359newline-separated segment count, which includes the trailing empty segment. Python splitlines correctly returns1358physical lines. No unread article text or semantic/source/runtime mismatch.",
        "correction": "Changed only the evidence line-count assertion and exact reading-depth description to1358, then reran only the failed audit and remaining whitespace check. Existing passed validator/manifest/budget/vocabulary and fresh five-case HOST were not repeated.",
        "preservedOriginal": "host-audit-823.txt.gz",
        "successfulConfirmation": "host-audit-823-retry.txt.gz",
        "permissionDenial": False
    }
    e["limits"] = "Qualified bounded HOST/finite target-relative semantic checkpoint, not independent review, native transport/rendering/drawing, universal pool safety or phone acceptance. Every original822/inherited gate and limitation is preserved."
    write(E, e)

    print("Packaged complete HOST evidence with original-byte preservation and unchanged runtime.")


def staged():
    audit()
    e = load(E)
    for record in e["provenance"] + e["tests"] + e["compressedEvidence"]:
        blob = subprocess.check_output(["git", "show", ":" + record["archive"]], cwd=R)
        assert hashlib.sha256(blob).hexdigest() == record["sha256"], record["archive"]
        if "rawSha256" in record:
            assert hashlib.sha256(gzip.decompress(blob)).hexdigest() == record["rawSha256"]
    print("Every staged evidence blob hash verified.")
