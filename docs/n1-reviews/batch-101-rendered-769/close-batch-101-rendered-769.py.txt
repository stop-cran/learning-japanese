def runtime():
    sys.path.insert(0, str(R / "tools"))
    from build_manifest import build_manifest, sha256

    manifest = load(R / "manifest.json")
    assert build_manifest(R) == manifest
    count = 0
    for kind, entries in manifest.items():
        if not isinstance(entries, dict):
            continue
        for name, digest in entries.items():
            path = R.joinpath(*name.split("/"))
            actual = sha(path) if kind in {
                "images", "audio", "listeningTaskAudio", "listeningTaskImages"
            } else sha256(path)
            assert actual == digest, name
            count += 1
    assert count == 8455 and manifest["contentVersion"] == "6e9b74967e21fc3c"
    assert not git("diff", review.TEACHING, "HEAD", "--",
                   "kanji", "words", "articles", "strokes", "manifest.json", "tools")
    return {
        "runtimeHashesVerified": count, "contentVersion": manifest["contentVersion"],
        "manifestSha256": sha(R / "manifest.json"), "manifestFresh": True,
        "teachingAndAllRuntimeBytesUnchanged": True,
    }


def records():
    result = {}
    for label, (model, session) in review.SESSIONS.items():
        record = load(C / f"batch-101-rendered-769-{label}-execution.json")
        prefix = C / f"batch-101-rendered-769-{label}"
        assert record["actualModels"] == [model] and record["usageCurrentModel"] == model
        assert record["sessionId"] == session and record["exitCode"] == 0
        assert review.absent(record["launcherPid"]) and record["actualProcessExitVerified"]
        assert record["allAssignedResourcesSuccessfullyReturned"]
        assert not record["unfinishedTools"] and not record["observedToolOverlaps"]
        assert record["exactlyOneViewPerToolBearingMessage"]
        assert len(record["actualDecodedImageReturns"]) == 26
        assert all(call["name"] == "view" and call["success"] for call in record["toolCalls"])
        assert record["rawSha256"] == sha(prefix.with_suffix(".jsonl"))
        assert record["reportSha256"] == sha(prefix.with_suffix(".md"))
        result[label] = record
    assert result["claude"]["completedAt"] < result["gpt"]["startedAt"]
    assert [i["returnedSha256"] for i in result["claude"]["actualDecodedImageReturns"]] == [
        i["returnedSha256"] for i in result["gpt"]["actualDecodedImageReturns"]]
    return result
