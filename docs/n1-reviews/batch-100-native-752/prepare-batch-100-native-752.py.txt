def load(p):
    return json.loads(p.read_text(encoding="utf-8-sig"))


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def save(p, value):
    assert not p.exists(), p
    p.write_text(json.dumps(value, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(p) == value


def git(*args, root=R):
    return subprocess.check_output(["git", "--no-pager", "-C", str(root), *args], encoding="utf-8").strip()


def context(clean_content=True):
    p = load(C / "progress.json")
    assert (p["checkpoint"], p["batch"], p["status"]) == (751, "100", "continue")
    runner = load(C / "runner.json")
    assert (runner["status"], runner["sessionId"], runner["pid"]) == ("running", SESSION, 5920)
    for root, head in [(R, BASE), (A, APP)]:
        assert git("rev-parse", "HEAD", root=root) == head
        assert git("log", "-1", "--format=%G?", root=root) == "G"
        if root == A or clean_content:
            assert not git("status", "--porcelain", "--untracked-files=all", root=root)
    return p


def replace_progress(before, after, phase):
    tmp = C / f"progress.json.752-{phase}.tmp"
    save(tmp, after)
    assert load(C / "progress.json") == before
    os.replace(tmp, C / "progress.json")
    assert load(C / "progress.json") == after

def plain(text):
    text = re.sub(r"\[([^\]]+)\]\([^)]+\)", r"\1", text)
    return text.replace("**", "").replace("`", "")


def kstr(text):
    return json.dumps(text, ensure_ascii=False)


def klist(values):
    return "listOf(" + ", ".join(map(kstr, values)) + ")"
