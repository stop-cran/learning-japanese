def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(path, value):
    assert not path.exists(), path
    path.write_text(json.dumps(value, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(path) == value


def git(*args, root=R):
    return subprocess.check_output(["git", "--no-pager", "-C", str(root), *args],
                                   encoding="utf-8").strip()


def context():
    p = load(C / "progress.json")
    assert (p["checkpoint"], p["batch"], p["status"]) == (788, "103", "continue")
    runner = load(C / "runner.json")
    assert (runner["status"], runner["pid"], runner["sessionId"]) == ("running", 5920, L.name)
    assert git("rev-parse", "HEAD", root=A) == APP
    assert git("log", "-1", "--format=%G?", root=A) == "G"
    assert not git("status", "--porcelain", "--untracked-files=all", root=A)
    return p


def replace_progress(current, after, phase):
    temporary = C / f"progress.json.789-{phase}.tmp"
    save(temporary, after)
    assert load(C / "progress.json") == current
    os.replace(temporary, C / "progress.json")
    assert load(C / "progress.json") == after


def runtime():
    manifest = load(R / "manifest.json")
    binary = {"images", "audio", "listeningTaskAudio", "listeningTaskImages"}
    hashes = {}
    for section_name, section in manifest.items():
        if isinstance(section, dict):
            for name, digest in section.items():
                assert name not in hashes
                data = (R / name).read_bytes()
                if section_name not in binary:
                    data = data.replace(b"\r\n", b"\n")
                assert hashlib.sha256(data).hexdigest() == digest, name
                hashes[name] = digest
    assert len(hashes) == 8463 and manifest["contentVersion"] == "38c32a8df5fd6063"
    return {"contentVersion": manifest["contentVersion"], "runtimeHashesVerified": len(hashes),
            "manifestSha256": sha(R / "manifest.json")}
