def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(path, value):
    assert not path.exists(), path
    path.write_text(json.dumps(value, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(path) == value


def git(*args, root=R):
    return subprocess.check_output(["git", "--no-pager", "-C", str(root), *args], encoding="utf-8").strip()


def context(clean_content=True):
    progress = load(C / "progress.json")
    assert (progress["checkpoint"], progress["batch"], progress["status"]) == (756, "100", "continue")
    runner = load(C / "runner.json")
    assert (runner["status"], runner["sessionId"], runner["pid"]) == ("running", SESSION, 5920)
    for root, head in [(R, BASE), (A, APP)]:
        assert git("rev-parse", "HEAD", root=root) == head
        assert git("log", "-1", "--format=%G?", root=root) == "G"
        if root == A or clean_content:
            assert not git("status", "--porcelain", "--untracked-files=all", root=root)
    return progress


def replace_progress(before, after, phase):
    temporary = C / f"progress.json.757-{phase}.tmp"
    save(temporary, after)
    assert load(C / "progress.json") == before
    os.replace(temporary, C / "progress.json")
    assert load(C / "progress.json") == after

def replace_exact(text, before, after):
    assert text.count(before) == 1, before
    return text.replace(before, after)
