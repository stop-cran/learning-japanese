def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def save(path, value):
    assert not path.exists(), path
    path.write_text(json.dumps(value, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(path) == value


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git(*args, root=ROOT):
    return subprocess.check_output(
        ["git", "--no-pager", "-C", str(root), *args], encoding="utf-8"
    ).strip()

def runtime():
    packet = load(CONTROL / "batch-100-native-757" / "compiled-packet.json")
    prep = packet["preparation"]
    for path, digest in [
        (prep["archive"], prep["archiveSha256"]),
        (prep["fixture"], prep["fixtureSha256"]),
        (packet["compilation"]["appApk"], packet["compilation"]["appApkSha256"]),
        (packet["compilation"]["testApk"], packet["compilation"]["testApkSha256"]),
    ]:
        assert sha(Path(path)) == digest
    manifest = load(ROOT / "manifest.json")
    assert manifest["contentVersion"] == "606c836e4be33ba9"
    assert sha(ROOT / "manifest.json") == "c11e9f588a0a487579a5d0adac2f35cf90523fc412d6444eb2af9e9bab56df73"
    archive_path = Path(prep["archive"])
    with zipfile.ZipFile(archive_path) as archive:
        names = archive.namelist()
        assert len(names) == len(set(names)) == 8475 and len(names) <= 18000
        assert all(n.startswith("n1-batch-100/") and "\\" not in n and ".." not in n.split("/")
                   and not n.startswith("/") and not re.search(r"(^|/)[^/]*:", n) for n in names)
        assert not any("/docs/" in n or "/tools/" in n for n in names)
        assert archive.read("n1-batch-100/manifest.json") == (ROOT / "manifest.json").read_bytes()
        retained = len((ROOT / "manifest.json").read_bytes())
        count = media = 0
        for section_name, section in manifest.items():
            if not isinstance(section, dict):
                continue
            for name, digest in section.items():
                assert len(name.split("/")) == 2 and ".." not in name.split("/")
                data = archive.read("n1-batch-100/" + name)
                local = (ROOT / name).read_bytes()
                binary = section_name in {"images", "audio", "listeningTaskAudio", "listeningTaskImages"}
                limit = 150 * 1024 if section_name in {"images", "listeningTaskImages"} else (
                    512 * 1024 if section_name in {"audio", "listeningTaskAudio"} else 2 * 1024 * 1024)
                assert len(data) <= limit, name
                normalized = data if binary else data.replace(b"\r\n", b"\n")
                assert hashlib.sha256(normalized).hexdigest() == digest, name
                assert normalized == (local if binary else local.replace(b"\r\n", b"\n")), name
                retained += len(data)
                count += 1
                if section_name in {"listeningTaskAudio", "listeningTaskImages"}:
                    media += len(data)
        inflated = sum(info.file_size for info in archive.infolist())
        assert count == 8453 and retained == 44381474 and inflated == 44649812
        assert retained <= 56 * 1024 * 1024 and inflated <= 80 * 1024 * 1024
        assert archive_path.stat().st_size == 30476969 <= 42 * 1024 * 1024
        assert media == 5893602 <= 20 * 1024 * 1024
    return {"runtimeHashesVerified": count, "contentVersion": manifest["contentVersion"],
            "manifestSha256": sha(ROOT / "manifest.json"), "zipEntries": len(names),
            "compressedBytes": archive_path.stat().st_size, "inflatedBytes": inflated,
            "retainedBytes": retained, "listeningTaskMediaBytes": media,
            "archiveSha256": prep["archiveSha256"], "allOriginalBudgetsAndPathMediaGuardsPassed": True}
