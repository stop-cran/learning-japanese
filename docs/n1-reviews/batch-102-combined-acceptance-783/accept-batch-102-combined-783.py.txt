def load(p):
    return json.loads(p.read_text(encoding="utf-8-sig"))


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def save(p, data, replace=False):
    assert replace or not p.exists(), p
    temp = p.with_suffix(p.suffix + ".783.tmp")
    assert not temp.exists(), temp
    temp.write_text(json.dumps(data, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(temp) == data
    os.replace(temp, p)
    assert load(p) == data


def git(*args, root=R):
    return subprocess.check_output(
        ["git", "--no-pager", "-c", "core.quotepath=false", "-C", str(root), *args],
        encoding="utf-8").strip()

def canonical(e):
    return e.tag, tuple(sorted(e.attrib.items())), (e.text or "").strip(), tuple(canonical(c) for c in e)

def indexed(folder):
    count = 0
    for row in load(folder / "artifact-index.json"):
        path = folder / row["file"]
        raw = path.read_bytes()
        assert sha(raw) == row["storedSha256"], path
        decoded = gzip.decompress(raw) if path.suffix == ".gz" else raw
        assert sha(decoded) == row["sourceSha256"] and len(decoded) == row["sourceBytes"]
        count += 1
    return count
