def load(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(path, value):
    assert not path.exists(), path
    path.write_text(json.dumps(value, ensure_ascii=True, indent=2) + "\n", encoding="utf-8")
    assert load(path) == value

def runtime():
    manifest = load(R / "manifest.json")
    binary = {"images", "audio", "listeningTaskAudio", "listeningTaskImages"}
    hashes = {}
    for section, entries in manifest.items():
        if not isinstance(entries, dict):
            continue
        for name, digest in entries.items():
            assert name not in hashes
            data = (R / name).read_bytes()
            normalized = data if section in binary else data.replace(b"\r\n", b"\n")
            assert hashlib.sha256(normalized).hexdigest() == digest, name
            hashes[name] = digest
    assert len(hashes) == 8458 and manifest["contentVersion"] == "f376333b242ebb23"
    return {"contentVersion": manifest["contentVersion"], "runtimeHashesVerified": len(hashes),
            "manifestSha256": sha(R / "manifest.json")}

def plain(text):
    return re.sub(r"\[([^\]]+)\]\([^)]+\)", r"\1", text).replace("**", "").replace("`", "")


def kstr(text):
    return json.dumps(text, ensure_ascii=False)


def klist(values):
    return "listOf(" + ", ".join(map(kstr, values)) + ")"


def article_sections(card, prefix):
    result = []
    for block in card.split("---", 2)[2].strip().split("\n\n"):
        if block.startswith("# "):
            continue
        if block.startswith("## "):
            result.append(f"            section({kstr(block[3:].strip())})")
            continue
        for line in (block.splitlines() if block.startswith("|") or block.startswith("- ") else [block]):
            if line.startswith("|"):
                cells = [cell.strip() for cell in line.strip("|").split("|")]
                if cells[0] == "Reading" or cells[0].startswith("---"):
                    continue
                for cell in (cells[0], cells[2]):
                    result.append(f"            show(readingExample({kstr(plain(cell))}), \"{prefix}-{len(result):02d}\")")
            else:
                text = plain(" ".join(line.removeprefix("- ").splitlines()))
                result.append(f"            show(articleBullet({kstr(text)}), \"{prefix}-{len(result):02d}\")")
    return "\n".join(result)
